  <script>
    // -- View Switching Logic --
    function switchView(viewId) {
      // Hide all views
      document.querySelectorAll('.view').forEach(el => el.classList.remove('active'));
      // Show selected view
      document.getElementById('view-' + viewId).classList.add('active');
      
      // Update sidebar active states
      document.querySelectorAll('.nav-item').forEach(el => el.classList.remove('active'));
      const activeNav = document.getElementById('nav-' + viewId);
      if(activeNav) activeNav.classList.add('active');
    }

    // -- State Arrays to hold dynamic data --
    let projects = [
      { name: "water", desc: "No description provided.", date: new Date().toLocaleDateString() },
      { name: "Math", desc: "working on school", date: new Date().toLocaleDateString() }
    ];

    let ions = [];

    // -- Render Functions --
    function renderProjects() {
      const grid = document.getElementById('projects-grid');
      grid.innerHTML = ''; // Clear current
      
      projects.forEach(proj => {
        grid.innerHTML += `
          <div class="card">
            <div class="card-icon-wrapper"><span class="material-symbols-outlined">folder</span></div>
            <h3 class="card-title">${proj.name}</h3>
            <p class="card-desc">${proj.desc}</p>
            <div class="card-footer">
              <span>INITIALIZED: ${proj.date}</span>
              <span class="card-link">Open Workspace →</span>
            </div>
          </div>
        `;
      });
    }

    function renderIons() {
      const grid = document.getElementById('ions-grid');
      // Keep the "Add Card" as the first element
      grid.innerHTML = `
        <div class="card add-card" onclick="addIon()">
          <span class="material-symbols-outlined" style="font-size: 32px; margin-bottom: 10px;">add_circle</span>
          <div style="font-weight: 600;">NEW NEURAL ION</div>
        </div>
      `;
      
      ions.forEach(ion => {
        grid.innerHTML += `
          <div class="card">
            <div class="card-icon-wrapper" style="color: ${ion.color}; background: ${ion.color}20;"><span class="material-symbols-outlined">smart_toy</span></div>
            <h3 class="card-title">${ion.name}</h3>
            <p class="card-desc">${ion.instruction}</p>
            <div class="card-footer">
              <span style="color: ${ion.color}">● Active Sync</span>
              <span class="card-link">Configure →</span>
            </div>
          </div>
        `;
      });
    }

    // -- Action Handlers --
    function addProject() {
      const name = prompt("Enter project name:");
      if (!name) return;
      const desc = prompt("Enter project description (optional):") || "No description provided.";
      projects.push({ name, desc, date: new Date().toLocaleDateString() });
      renderProjects();
    }

    function addIon() {
      const name = prompt("Enter Neural Ion Designation (Name):");
      if (!name) return;
      const instruction = prompt("Enter core logical instruction for this Ion:") || "Standard logical processing sequence.";
      const colors = ['#007bff', '#ff4d4d', '#00e676', '#a200ff', '#ff9900'];
      const randomColor = colors[Math.floor(Math.random() * colors.length)];
      
      ions.push({ name, instruction, color: randomColor });
      renderIons();
    }

    // Initialize the UI on load
    renderProjects();
    renderIons();
  </script>
